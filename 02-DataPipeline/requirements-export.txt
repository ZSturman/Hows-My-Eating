# Pinned dependency set for the full iteration loop
# (record/correct in the app -> import field tests -> retrain -> deploy CoreML).
#
# Why these pins?
# - torch 2.9.x + coremltools >=8 segfaults inside ct.convert on macOS arm64 as
#   of this writing. torch 2.7.x is the most recent torch coremltools has been
#   tested against, so we pin to that for the export step. Training works fine
#   on the same torch, so a single venv covers the whole loop.
# - coremltools 9 is shipped on PyPI but the convert path is unstable with
#   newer torches; pin to the 8.x series until that catches up.
#
# Usage:
#   python3 -m venv .venv-export
#   source .venv-export/bin/activate
#   pip install -r 02-DataPipeline/requirements-export.txt
#
# Then run:
#   python main_new.py from-field-test --input data/user/field_tests
#   python main_new.py compare
#   python main_new.py deploy --force

# --- core math / tensors -----------------------------------------------------
torch==2.7.*
numpy>=1.26,<2.3
scipy>=1.11,<1.16
scikit-learn>=1.3,<1.6

# --- data wrangling ----------------------------------------------------------
pandas>=2.1,<2.4
PyYAML>=6.0
tqdm>=4.66

# --- coreml export -----------------------------------------------------------
coremltools>=8.1,<9

# --- experiment tracking (optional but referenced by train.py) --------------
wandb>=0.18.0
